  function makeImage() {
    const img = new Image();
    img.src = imageUrl;
    img.alt = '';
    img.draggable = false;
    return img;
  }

  function buildScene() {
    flat.appendChild(makeImage());

    for (let g=0; g<4; g++) {
      const ghost = document.createElement('div');
      ghost.className = 'ghost';
      ghost.appendChild(makeImage());
      ghostsEl.appendChild(ghost);
      ghostModels.push({el:ghost, i:g});
    }

    for (let i=0; i<cuts.length-1; i++) {
      const top = cuts[i];
      const bottom = cuts[i+1];
      const h = bottom-top;
      const slice = document.createElement('div');
      slice.className = 'slice';
      slice.style.setProperty('--slice-top', `${top*100}%`);
      slice.style.setProperty('--slice-h', `${h*100}%`);
      slice.style.setProperty('--image-offset', `${-(top/h)*100}%`);
      const front = document.createElement('div');
      front.className = 'front';
      front.appendChild(makeImage());
      slice.appendChild(front);
      slicesEl.appendChild(slice);
      sliceModels.push({el:slice, i, top, h, mid:(top+bottom)/2});
    }
  }

  function sizeScene() {
    const vh = innerHeight;
    const vw = innerWidth;
    const maxH = vw < 700 ? vh*.68 : vh*.76;
    const maxW = vw < 700 ? vw*.76 : vw*.49;
    const width = Math.min(maxW, maxH*(SOURCE_W/SOURCE_H));
    const height = width*(SOURCE_H/SOURCE_W);
    root.style.setProperty('--score-w', `${width}px`);
    root.style.setProperty('--score-h', `${height}px`);
    root.style.setProperty('--edge', `${clamp(width*.025, 8, 16)}px`);
  }

  function targetProgress() {
    if (Number.isFinite(debugProgress)) return clamp(debugProgress);
    const max = Math.max(1, track.offsetHeight - innerHeight);
    const rect = track.getBoundingClientRect();
    return clamp(-rect.top/max);
  }

  function setPhase(p) {
    let txt = '01 / surface';
    if (p >= .12) txt = '02 / score becomes structure';
    if (p >= .34) txt = '03 / staves become planes';
    if (p >= .58) txt = '04 / orbit';
    if (p >= .79) txt = '05 / compression';
    if (p >= .93) txt = '06 / surface again';
    phaseEl.textContent = txt;
    scrollNote.style.opacity = String(1-smooth(.02,.11,p));
  }

  function render(p) {
    root.style.setProperty('--progress', p.toFixed(4));
    setPhase(p);

    const open = reduced ? 0 : bell(p, .08, .22, .80, .95);
    const explode = reduced ? 0 : bell(p, .18, .34, .72, .91);
    const orbitA = reduced ? 0 : smooth(.10,.55,p) * (1-smooth(.84,.98,p));
    const orbitB = reduced ? 0 : smooth(.56,.72,p) * (1-smooth(.84,.97,p));
    const returnFlat = smooth(.90,.985,p);
    const introScale = mix(.93, 1.0, smooth(0,.12,p));
    const depthScale = mix(1, 1.10, bell(p,.34,.48,.73,.86));

    const rotY = (-41*orbitA + 64*orbitB) * (1-returnFlat) + pointerX*1.6*open;
    const rotX = (8*orbitA - 5*orbitB) * (1-returnFlat) - pointerY*1.25*open;
    const rotZ = (1.2*Math.sin(p*Math.PI*3))*open;
    const x = mix(0, innerWidth < 700 ? -5 : -22, bell(p,.48,.58,.77,.90));
    const y = mix(10, -8, smooth(.12,.75,p));
    world.style.transform = `translate3d(${x}px,${y}px,0) rotateX(${rotX}deg) rotateY(${rotY}deg) rotateZ(${rotZ}deg) scale(${introScale*depthScale})`;

    const flatAlpha = 1 - smooth(.075,.18,p) + returnFlat;
    flat.style.opacity = String(clamp(flatAlpha));
    flat.style.transform = `translateZ(${mix(0,-8,open)}px)`;

    ghostModels.forEach(({el,i}) => {
      const stagger = i/3;
      const alpha = open * (1-returnFlat) * (.13 - i*.016);
      const gz = -48 - i*62 - explode*110*stagger;
      const gy = (i-1.5)*10*explode;
      const grY = rotY*.18 + (i-1.5)*1.8*explode;
      el.style.opacity = String(clamp(alpha));
      el.style.transform = `translate3d(0,${gy}px,${gz}px) rotateY(${grY}deg)`;
    });

    const n = sliceModels.length-1;
    sliceModels.forEach((m) => {
      const centred = (m.i/n)*2-1;
      const wave = Math.sin((m.i*1.71)+.4);
      const alt = m.i%2 ? 1 : -1;
      const envelope = open*(1-returnFlat);

      const zBase = explode * (130 + 560*(m.i/n));
      const zWave = explode * wave*52;
      const xFan = explode * centred * (innerWidth<700 ? 42 : 86);
      const xWave = explode * wave * (innerWidth<700 ? 12 : 24);
      const yFan = explode * centred * centred * alt * 28;
      const localRY = explode * centred * 8 + orbitB*centred*4;
      const localRX = explode * wave*2.6;
      const localRZ = explode * alt*centred*1.4;

      const z = envelope*(zBase+zWave);
      const sx = envelope*(xFan+xWave);
      const sy = envelope*yFan;
      m.el.style.opacity = String(clamp(smooth(.075,.16,p) * (1-returnFlat*.86)));
      m.el.style.transform = `translate3d(${sx}px,${sy}px,${z}px) rotateX(${localRX}deg) rotateY(${localRY}deg) rotateZ(${localRZ}deg)`;
      m.el.style.filter = `brightness(${mix(1, .94, explode*Math.max(0, centred))})`;
    });

    const shadow = document.getElementById('shadow');
    shadow.style.opacity = String(.22 + open*.32);
    shadow.style.transform = `translate3d(${rotY*1.2}px, ${34+explode*24}px, ${-170-explode*160}px) rotateX(74deg) scale(${1+explode*.18})`;
  }

  function loop() {
    raf = requestAnimationFrame(loop);
    scrollProgress = targetProgress();
    const follow = Number.isFinite(debugProgress) ? 1 : .13;
    visualProgress += (scrollProgress-visualProgress)*follow;
    if (Math.abs(scrollProgress-visualProgress) < .00005) visualProgress = scrollProgress;
    render(visualProgress);
  }

  function onPointer(e) {
    if (e.pointerType === 'touch') return;
    pointerX = ((e.clientX/innerWidth)-.5)*2;
    pointerY = ((e.clientY/innerHeight)-.5)*2;
    root.style.setProperty('--pointer-x', pointerX.toFixed(3));
    root.style.setProperty('--pointer-y', pointerY.toFixed(3));
  }

  function cleanup() {
    cancelAnimationFrame(raf);
    if (imageUrl) URL.revokeObjectURL(imageUrl);
  }

  addEventListener('resize', sizeScene, {passive:true});
  addEventListener('pointermove', onPointer, {passive:true});
  addEventListener('pagehide', cleanup, {once:true});

  loadImage()
    .then(() => {
      buildScene();
      sizeScene();
      loading.hidden = true;
      visualProgress = targetProgress();
      render(visualProgress);
      loop();
    })
    .catch(err => {
      loading.querySelector('span').textContent = 'score unavailable';
      console.error(err);
    });
})();
